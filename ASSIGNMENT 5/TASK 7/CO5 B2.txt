
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

int main(void)
{
    char name[20];

    printf("Enter donor name: ");

    if (fgets(name, sizeof name, stdin) == NULL)
    {
        perror("fgets");
        return 1;
    }

    name[strcspn(name, "\n")] = '\0';

    if (strlen(name) == sizeof name - 1)
    {
        printf("Warning: input may be truncated\n");
    }

    printf("Name safely read: %s\n", name);

    FILE *fp = fopen("missing_donors.txt", "r");

    if (fp == NULL)
        perror("missing_donors.txt");
    else
        fclose(fp);

    return 0;
}
