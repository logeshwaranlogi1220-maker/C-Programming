#include <stdio.h>
char grade(int marks)
{
    if (marks < 0 || marks > 100) return '?';
    switch (marks / 10) {
        case 10: case 9: return 'S';
        case 8: return 'A';
        case 7: return 'B';
        case 6: return 'C';
        case 5: return 'D';
        default: return 'F';
    }
}

char grade_ladder(int m)
{
    if (m < 0 || m > 100) return '?';
    else if (m >= 90) return 'S'; else if (m >= 80) return 'A'; else if (m >= 70) return 'B';
    else if (m >= 60) return 'C'; else if (m >= 50) return 'D'; else return 'F';
}

char grades_tab[11] = {'F','F','F','F','F','D','C','B','A','S','S'};
char grade_table(int m) { return (m < 0 || m > 100) ? '?' : grades_tab[m / 10]; }

int main(void)
{
    int marks[] = {100, 95, 83, 50, 49, -5};
    for (int i = 0; i < 6; i++) printf("%d:%c ", marks[i], grade(marks[i]));
    printf("\n");
    return 0;
}
