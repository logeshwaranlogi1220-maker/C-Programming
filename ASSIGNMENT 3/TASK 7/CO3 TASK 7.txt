#include <stdio.h>

#define MAX_CANDIDATES 500
#define MAX_VOTES 500000

int main(void)
{
    int count[MAX_CANDIDATES + 1] = {0};
    int order[MAX_CANDIDATES];

    int c, v, valid = 0, invalid = 0;

    printf("Number of candidates (1-%d): ", MAX_CANDIDATES);

    if (scanf("%d", &c) != 1 ||
        c < 1 || c > MAX_CANDIDATES) {
        fprintf(stderr, "Invalid candidate count.\n");
        return 1;
    }

    printf("Number of votes (0-%d): ", MAX_VOTES);

    if (scanf("%d", &v) != 1 ||
        v < 0 || v > MAX_VOTES) {
        fprintf(stderr, "Invalid vote count.\n");
        return 1;
    }

    for (int i = 1; i <= c; i++)
        order[i - 1] = i;

    for (int i = 0; i < v; i++) {
        int vote;

        if (scanf("%d", &vote) != 1) {
            fprintf(stderr, "Invalid input while reading votes.\n");
            return 1;
        }

        if (vote >= 1 && vote <= c) {
            count[vote]++;
            valid++;
        } else {
            invalid++;
        }
    }

    /* Sort candidate IDs by descending vote count */

    for (int i = 0; i < c - 1; i++) {
        int best = i;

        for (int j = i + 1; j < c; j++) {
            if (count[order[j]] > count[order[best]])
                best = j;
        }

        if (best != i) {
            int t = order[i];
            order[i] = order[best];
            order[best] = t;
        }
    }

    printf("\nValid votes: %d\n", valid);
    printf("Invalid votes: %d\n", invalid);

    printf("Ranking (candidate: votes):\n");

    for (int i = 0; i < c; i++)
        printf("%d: %d\n", order[i], count[order[i]]);

    int maximum = count[order[0]];
    int winners = 0;

    for (int i = 0; i < c; i++) {
        if (count[order[i]] == maximum)
            winners++;
    }

    if (winners > 1) {
        printf("Result: TIE among");

        for (int i = 0; i < c; i++) {
            if (count[order[i]] == maximum)
                printf(" candidate %d", order[i]);
        }

        putchar('\n');
    } else {
        printf("Winner: Candidate %d\n", order[0]);
    }

    return 0;
}
